pub struct Plan {
    pub net:      NetChoice,
    pub locale:   Locale,             // language, keyboard, time zone
    pub distro:   Distro,             // Arch | Debian | Fedora | OpenSuse
    pub desktop:  Option<Desktop>,    // None = "No desktop", skips 04
    pub env:      &'static str,       // minimal | server | workstation…
    pub addons:   Vec<&'static str>,
    pub picks:    Picks,              // ticks and unticks inside groups
    pub apps:     Selection,
    pub prefer:   Prefer,             // distro packages or Flathub first
    pub sources:  Sources,            // driver sources locked on
    pub disk:     DiskPlan,           // Alongside{size} | Whole
    pub user:     Identity,           // one passphrase, or a separate admin
}

review → install · the only door
pub enum Step { Network, Locale, Distro, Desktop, System,
                Disk, You, Review, Install, Done, Recover }

(Review, Msg::Install) => {
    let job = Job::from(&m.plan, &m.hw);  // pure
    m.step = Install;
    Cmd::Run(job)                         // first write
}
(Install, Failed(f)) => { m.fault = Some(f); m.step = Recover; }
(s, Back) => m.step = s.prev(&m.plan), // skips Desktop
