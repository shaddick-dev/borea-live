borea-core · update 
#[derive(Clone, Copy, PartialEq)]
pub enum Step { Network, Locale, Compose, Disk, Identity,
                Review, Install, Done, Recover }

pub enum Msg  { Key(Key), Probed(Probe), Net(NetState),
                Progress(JobEvent), Failed(Fault), Tick }

pub enum Cmd  { None, Probe, JoinWifi(Creds), Run(Job), Reboot }

pub fn update(m: &mut Model, msg: Msg) -> Cmd {
    use {Msg::*, Step::*};
    match (m.step, msg) {
        (Review, Key(Key::HoldEnter)) => {
            let job = Job::from(render::all(&m.plan));  // pure
            m.step = Install;
            Cmd::Run(job)                                // first write
        }
        (Install, Progress(ev)) => { m.job.apply(ev); Cmd::None }
        (Install, Failed(f))    => { m.fault = Some(f); m.step = Recover; Cmd::None }
        (Install, Key(Key::Esc)) => Cmd::None,         // locked
        (s, Key(Key::Esc)) => { m.step = s.prev(); Cmd::None }
        (s, Key(Key::Enter)) => match m.plan.validate(s) {
            Ok(())    => { m.step = s.next(); Cmd::None }
            Err(hint) => { m.hint = Some(hint); Cmd::None }
        },
        (s, Key(k)) => { screens::handle(s, k, &mut m.plan); Cmd::None }
        (_, Probed(p)) => { m.probe = p; m.plan.suggest(&m.probe); Cmd::None }
        (_, Tick) => { m.anim.step(); Cmd::None }    // 80 ms clock
        _ => Cmd::None,
    }
}
